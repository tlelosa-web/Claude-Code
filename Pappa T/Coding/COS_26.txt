        function submitTask() {
            const category = document.getElementById('category').value;
            const priority = document.getElementById('priority').value;
            const task = document.getElementById('task').value;
            const context = document.getElementById('context').value;

            if (!task.trim()) {
                alert('Please enter a task description');
                return;
            }

            const output = document.getElementById('output');
            const timestamp = new Date().toLocaleString();
            
            output.innerHTML = `
                <div class="timestamp">${timestamp}</div>
                <div style="color: var(--accent); font-weight: 600; margin-bottom: 1rem;">TASK SUBMITTED</div>
                <div><strong>Category:</strong> ${category.toUpperCase()}</div>
                <div><strong>Priority:</strong> ${priority.toUpperCase()}</div>
                <div style="margin-top: 1rem;"><strong>Description:</strong></div>
                <div style="margin-top: 0.5rem; padding: 1rem; background: var(--secondary); border-radius: 6px;">${task}</div>
                ${context ? `<div style="margin-top: 1rem;"><strong>Context:</strong></div><div style="margin-top: 0.5rem; padding: 1rem; background: var(--secondary); border-radius: 6px;">${context}</div>` : ''}
                <div style="margin-top: 1.5rem; padding: 1rem; background: rgba(0, 212, 170, 0.1); border-left: 3px solid var(--accent); border-radius: 4px;">
                    <strong>→ Rudy (Chief of Staff) will process this task according to Charter protocols.</strong><br><br>
                    Please continue this conversation in the main chat to receive:
                    <ul style="margin-left: 1.5rem; margin-top: 0.5rem;">
                        <li>Analysis and options</li>
                        <li>Risk assessment if applicable</li>
                        <li>Action plan with timelines</li>
                        <li>Resource requirements</li>
                    </ul>
                </div>
            `;
            output.classList.add('active');

            // Scroll to output
            output.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
        }
